\section{Preliminaries}
\noindent\textbf{Notions}. We use $\lambda$ for the security parameter, and $\mathsf{negl}(\lambda)$ for the negligible function, i.e., for any polynomial $f$, there exists a $\lambda_0$ such that for all $\lambda>\lambda_0,\mathsf{negl}(\lambda)<1/f(\lambda)$. We use $[n]$ for the set $\{1,2,\dots,n\}$. For a set $S$, we use $S'$ for $S\cup\{0\}$, and use $[n]'$ for $[n]\cup\{0\}=\{0,1,\dots,n\}$.

\subsection{Bilinear Groups}
We use $(\mathbb{Z}_p,\mathbb{G}_1,\mathbb{G}_2,\mathbb{G}_T,e,g_1,g_2,g_T)$ for a pairing-friendly group, where $\mathbb{Z}_p$ is a prime field of order $p=\exp(\Omega(\lambda))$, and $\mathbb{G}_1,\mathbb{G}_2,\mathbb{G}_T$ are groups of order $p$ with a non-degenerate pairing $e:\mathbb{G}_1\times\mathbb{G}_2\to\mathbb{G}_T$. $g_1\in\mathbb{G}_1,g_2\in\mathbb{G}_2$ are uniformly sampled generators such that $e(g_1,g_2)=g_T$. For $x\in\mathbb{Z}_p$, we use $[x]_s$ to denote $x\cdot g_s$, and we denote the group operation additively, namely $[x]_s+[y]_s=[x+y]_s$ for $s\in\{1,2,T\}$. We denote the pairing operation $e([x]_1,[y]_2)$ by $[x]_1\circ[y]_2=[y]_2\circ[x]_1=[x\cdot y]_T$. We also use this notion for matrices of group elements, i.e., for a $k_1\times k_2$ matrix $\bm{A}=\{[a_{i,k}]_{\alpha_k}\}$ and a $k_2\times k_3$ matrix $\bm{B}=\{[b_{k,j}]_{\beta_k}\}$, if $\alpha_k+\beta_k=3$ for all $k=1,2,\dots,k_2$, then $\bm{A}\circ\bm{B}$ is well-defined, and its result is a $k_1\times k_3$ matrix $\bm{C}=\{[c_{i,j}]_T\}$ that satisfies $[c_{i,j}]_T=\sum_{k=1}^{k_2}[a_{i,k}]_{\alpha_k}\circ[b_{k,j}]_{\beta_k}$. It is obvious that $c_{i,j}=\sum_{k=1}^{k_2}a_{i,k}b_{k,j}$.

\subsection{BLS Signature}
Let $\mathcal{H}_2:\{0,1\}^*\to\mathbb{G}_2$ be a random oracle. The BLS signature scheme~\cite{boneh2001short} consists of the following algorithms:
\begin{itemize}
    \item $\mathsf{BLS.Gen}$: It samples a random $\mathsf{sk}\gets\mathbb{Z}_p$ and outputs a public/secret key pair as $(\mathsf{pk}=[\mathsf{sk}]_1,\mathsf{sk})$.
    \item $\mathsf{BLS.Sign}(\mathsf{sk},\mathsf{msg})$: It outputs a signature $\sigma=\mathsf{sk}\cdot\mathcal{H}_2(\mathsf{msg})\in\mathbb{G}_2$.
    \item $\mathsf{BLS.Verify}(\mathsf{pk},\mathsf{msg},\sigma)$: It verifies the validity of the signature by checking $\mathsf{pk}\circ\mathcal{H}_2(\mathsf{msg})\overset{?}{=}[1]_1\circ\sigma$.
\end{itemize}

\subsection{Equality of Discrete Logarithms Proofs} \label{subsec:eq-dl}
Let $\mathcal{H}_0:\{0,1\}^*\to\mathbb{Z}_p$ be a random oracle independent of $\mathcal{H}_2$. We use the Fiat–Shamir transform of the Chaum–Pedersen protocol~\cite{chaum1992wallet} to prove knowledge of a scalar $k$ such that $X=kA,Y=kB$, where $A,X\in\mathbb G_a$, $B,Y\in\mathbb G_b$, and $a,b\in\{1,2\}$. In our applications, $A=[1]_1$.

The Eq-DL protocol~\cite{kate2023flexirand} consists of the following algorithms:
\begin{itemize}
    \item $\mathsf{EqProve}(A,X,B,Y;k)$:
    \begin{enumerate}
        \item Sample $r\leftarrow_\$\mathbb{Z}_p$.
        \item Compute $\alpha=rA$ and $\beta=rB$.
        \item Compute $c=\mathcal{H}_0(A,X,B,Y,\alpha,\beta)$, and $s=r+ck$.
        \item Output $\pi=(c,s)$.
    \end{enumerate}
    \item $\mathsf{EqVrfy}(A,X,B,Y;\pi)$:
    \begin{enumerate}
        \item Reject malformed inputs and group elements of incorrect types.
        \item Parse $\pi=(c,s)$ and compute $\alpha=sA-cX, \beta=sB-cY$.
        \item Accept if and only if $c=\mathcal{H}_0(A,X,B,Y,\alpha,\beta)$.
    \end{enumerate}
\end{itemize}

This protocol is used to require users to prove knowledge of their $\mathsf{sk}$ during registration and partial signing (see Construction \ref{cons:c-sts-naive}).

\subsection{Polynomials over $\mathbb{Z}_p$}
Let $\omega\in\mathbb{Z}_p^*$ be an $M$-th primitive root of unity, and the multiplicative subgroup of roots of unity generated by $\omega$ is denoted by $\mathbb{H}=\{\omega,\omega^2,\dots,\omega^M\}$. Let $L_i(X)$ be the $i$-th Lagrange basis polynomial on $\mathbb{H}$, namely the unique degree $M-1$ polynomial which satisfies $L_i(\omega^j)$ is $1$ when $i=j$ and $0$ when $i\ne j$. Let $Z(X)=\prod_{i\in[M]}(X-\omega^i)$ be the vanishing polynomial on $\mathbb{H}$. Since $\mathbb{H}$ is a multiplicative subgroup, we have $Z(X)=X^M-1$, and $L_i(X)=\frac{\omega^i}{M}\cdot\frac{X^M-1}{X-\omega^i}$ with $L_i(0)=\frac{1}{M}$ for $i\in[M]$. Sometimes we also write $\omega^M$ as $\omega^0$ and $L_M$ as $L_0$ since they are equivalent.

For any vector $B=(b_1,b_2,\dots,b_M)$\footnote{If $B$ has length $l_B<M$, we consider that $b_i=0$ for all $i=l_B+1,\dots,M$.}, let $B(X)=\sum_{i\in[M]}b_i\cdot L_i(X)$ be the unique polynomial that satisfies $\deg(B)\le M-1$ and $B(\omega^i)=b_i$ for all $i\in[M]$.

\subsection{Polynomial Commitment}
We use the KZG polynomial commitment~\cite{kate2010constant} as the vector commitment scheme. Given a degree bound $d$, the common reference string derived by a trapdoor $\tau$ is denoted by $\mathsf{CRS}=([1],[\tau],[\tau^2],\dots,[\tau^d])$. For any polynomial $B(X)$ that satisfies $\deg(B)\le d$, its commitment is $[B(\tau)]$.

\subsection{Generalized Sumcheck}
Our construction relies on the following lemma, known as generalized sumcheck, whose proof can be found in~\cite{rafols2021algebraic}.
\begin{lemma}[Generalized Sumcheck] \label{lemma:sum}
Let $A(X)=\sum_{i=1}^{|\mathbb{H}|}a_i\cdot L_i(X),B(X)=\sum_{i=1}^{|\mathbb{H}|}b_i\cdot L_i(X)$, then
\begin{equation*}
    A(X)\cdot B(X)=\frac{\sum_ia_ib_i}{|\mathbb{H}|}+Q_x(X)\cdot X+Q_Z(X)\cdot Z(X)
\end{equation*}
where $Q_x,Q_Z$ are polynomials with degree $\leq |\mathbb{H}|-2$ defined as
\begin{equation*}
    \begin{aligned}
        Q_x(X)&=\sum_ia_ib_i\frac{L_i(X)-L_i(0)}{X}\\
        Q_Z(X)&=\sum_ia_ib_i\frac{L_i^2(X)-L_i(X)}{Z(X)}+\sum_{i\ne j}a_ib_j\frac{L_i(X)L_j(X)}{Z(X)}
    \end{aligned}
\end{equation*}
\end{lemma}

\section{Linear Verifiable c-STS Scheme} \label{sec:sts-naive}
In this section, we first recall the c-STS scheme Dyna-hinTS proposed in~\cite{kate2025dyna}, and discuss how to use the linear verification equations to replace the Plonk-style weight proofs.

\subsection{Revisiting Dyna-hinTS}
Dyna-hinTS is a silent threshold signature scheme for dynamic committees based on hinTS~\cite{garg2024hints}, but the authorized committee is a subset of the entire set of parties that changes dynamically over epochs. Assume there are $N$ parties in the system and $|\mathbb{H}|=N+1$. During the silent setup, each party will sample an independent BLS key pair $\{\mathsf{pk}_i=[\mathsf{sk}_i]_1,\mathsf{sk}_i\}$. At epoch $e$, the authorized committee $\mathsf{Com}_e$ is randomly chosen\footnote{Here we treat $\mathsf{Com}_e$ as a binary vector $\mathsf{Com}_e=\{b_1,\dots,b_N\}\in\{0,1\}^N$ where $b_i=1$ if and only if $i\in \mathsf{Com}_e$. The same applies to $B$ and $\bar{B}$, which we describe later.}. Given a message $m$ to sign, parties compute an epoch-specified\footnote{Embedding epoch $e$ into the signature prevents the signature from being used for other epochs.} $\mathsf{tag}=\mathcal{H}_2(m,e)$ and sign it using the BLS signature by $\sigma_i=\mathsf{sk}_i\cdot\mathsf{tag}$. Suppose a subset $B\subseteq[N]$ of parties have signed $\mathsf{tag}$; the aggregator will proceed to aggregate the partial signatures as follows. It first shall aggregate the partial signatures $\{\sigma_i\}_{i\in B}$ and public keys $\{\mathsf{pk}_i\}_{i\in B}$ as $\sigma^*$ and $\mathsf{aPK}$. Furthermore, to prove the honest aggregation of $\mathsf{aPK}$, the aggregator should commit to $B$ and perform the following computation:

\begin{enumerate}[label=\arabic*.]
    \item It computes proof $\pi_1=([Q_Z(\tau)]_1,[Q_x(\tau)]_1,[\hat{Q}_x(\tau)]_1)$ which relies on the generalized sumcheck protocol to prove that the inner-product of public keys and $B$ is the claimed aggregate public key $\mathsf{aPK}$:
    \begin{equation*}
        \begin{aligned}\
            [SK(\tau)]_1 \circ [B(\tau)]_2 &= [1]_2 \circ \mathsf{aPK} + [Z(\tau)]_2 \circ [Q_Z(\tau)]_1 \\&+ [\tau]_2 \circ [Q_x(\tau)]_1 \\
            [\tau]_2 \circ [Q_x(\tau)]_1 &= [1]_2 \circ [\hat{Q}_x(\tau)]_1 \\
        \end{aligned}
    \end{equation*}
    where $SK(X)=\sum_{i\in[N]}\mathsf{sk}_i\cdot L_i(X)$ and $B(X)=\sum_{i\in[N]}b_i\cdot L_i(X)$. The first equation checks that the polynomial identity holds at the random point $x=\tau$, and the second equation constrains the maximum degree of $Q_x(X)$ as required in the sumcheck protocol.
    
    \item It computes a Plonk-style proof $\pi_2$ to demonstrate that (1) $B$ is a binary vector, (2) the inner product between $B$ and $(1,1,\dots,1)$ equals $t$, and (3) $B$ is a subset of the authorized committee $\mathsf{Com}_e$. In particular, (3) is achieved in the same way as (1) and (2): By proving that $\bar{B}=\mathsf{Com}_e\setminus B$ is also a binary vector and the inner product between $\bar{B}$ and $(1,1,\dots,1)$ equals $|\mathsf{Com}_e|-t$, it is sufficient to show that $B\subseteq\mathsf{Com}_e$ (notice that given the commitment $[B_{\mathsf{Com}_e}(\tau)]_2$ of $\mathsf{Com}_e$, we have $[\bar{B}(\tau)]_2=[B_{\mathsf{Com}_e}(\tau)]_2-[B(\tau)]_2$).
\end{enumerate}

The final signature is $(\sigma^*,\mathsf{aPK},[B(\tau)]_2,\pi_1,\pi_2)$. For verification, one verifies $\pi_1,\pi_2$ and verifies the aggregated signature $\sigma^*$ by checking $\mathsf{tag} \circ \mathsf{aPK} = [1]_1 \circ \sigma^*$.

\subsection{Removing Plonk-style Proofs}
Since the computation of $[B(\tau)]_2,\mathsf{aPK},~[Q_Z(\tau)]_1,[Q_x(\tau)]_1,[\hat{Q}_x(\tau)]_1$ only relates to $i\in B$, it has complexity $O(n)$. However, generating Plonk-style proofs for proving $B$ and $\bar{B}$ have sufficient weight requires $O(N\log N)$ (below we use $B$, and the same also applies for $\bar{B}$) because it contains the following steps (slightly adapted and rephrased here for consistency with our notation; note that $L_{N+1}=L_0$):
\begin{itemize}
    \item Compute the partial sum vector $parsum$ defined by
    \begin{equation*}
        \begin{aligned}
            parsum_1&=0\\
            parsum_{i+1}&=parsum_i+b_i\qquad i=1,\dots,N,\\
        \end{aligned}
    \end{equation*}
    the corresponding polynomial
    \begin{equation*}
        Parsum(X)=\sum_{i=1}^{N+1}parsum_i\cdot L_i(X)
    \end{equation*}
    and its commitment $[Parsum(\tau)]_1$.
    \item To prove the correctness of $[Parsum(\tau)]_1$ and $[B(\tau)]_2$, i.e.,
    \begin{equation*}
        \begin{aligned}
            Parsum(X)&=0\qquad X=\omega\\
            Parsum(\omega X)-Parsum(X)-B(X)&=0\qquad X\in\mathbb{H}\setminus\{\omega^{N+1}\}\\
            Parsum(X)&=t\qquad X=\omega^{N+1}\\
            B(X)\cdot(1-B(X))&=0\qquad X\in\mathbb{H},
        \end{aligned}
    \end{equation*}
    compute the corresponding quotient polynomial $Q_i(X)$ for $i=1,2,3,4$ where
    \begin{equation*}
        \begin{aligned}
            L_1(X)\cdot Parsum(X)&=Q_1(X)\cdot Z(X)\\
            (X-\omega^{N+1})(Parsum(\omega X)-Parsum(X)-B(X))&=Q_2(X)\cdot Z(X)\\
            L_{N+1}(X)\cdot(Parsum(X)-t)&=Q_3(X)\cdot Z(X)\\
            B(X)\cdot(1-B(X))&=Q_4(X)\cdot Z(X)
        \end{aligned}
    \end{equation*}
    and their (linearly combined) commitment
    \begin{equation*}
        [Q(\tau)]_1=\Big[\sum_{i=1}^4Q_i(\tau)\cdot v^{i-1}\Big]_1
    \end{equation*}
    for some random challenge $v$.
    \item For some random challenge $r$, compute $B(r),Q(r),Parsum(r),\\Parsum(\omega r)$ with the corresponding KZG opening proofs.
\end{itemize}

Therefore, to reduce the complexity of aggregating signatures, we must avoid generating these proofs and use alternative methods to provide a proof of weight.~\cite{garg2024threshold} proposed a technique to achieve this goal: They compute $B(X)$ by interpolating $\{(i,0)\}_{[N]\setminus B}$, and a valid $B$ will satisfy $\deg(B)\le N-t$ which can be checked linearly via $[\tau^t]_1 \circ [B(\tau)]_2 = [1]_2 \circ [\hat{B}(\tau)]_1$ with an additional commitment of $\hat{B}(X)=B(X)\cdot X^t$. This works because $\mathsf{aPK}=\sum_{i=1}^n\mathsf{pk}_i\cdot b_i$ is an aggregation of sufficiently many ($\ge t$) public keys as long as this vector $B$ has $\ge t$ non-zero coordinates, even if $B$ is not binary. By the security of the BLS multi-signature, it is unforgeable if one does not have $\ge t$ partial signatures.

However, this trick cannot be used directly to replace $\pi_2$: $\pi_2$ proves that the weight of $B$ (resp. $\bar{B}$) \textbf{equals} $t$ (resp. $n-t$), and the degree check is limited to proving the weight $\ge t$. Even if it is used for proving the weight of $B\ge t$, the corresponding statement for $\bar{B}$ becomes "the weight of $\bar{B}\le n-t$" (not $\ge$), which is also beyond the ability of the degree check.

Our key observation for addressing this technical challenge is that \textbf{the vector $\mathsf{Com}_e$ does not have to be binary either}. One possible approach is to format $B_{\mathsf{Com}_e}(X)$ the same as $B(X)$, i.e., to compute $B_{\mathsf{Com}_e}(X)$ by interpolating $\{(i,0)\}_{[N]\setminus \mathsf{Com}_e}$. In this way, "$B\subseteq\mathsf{Com}_e$" is equivalent to "the zeros of $B_{\mathsf{Com}_e}(X)$ is a subset of the zeros of $B(X)$", and is further equivalent to the existence of the quotient polynomial $\bar{B}(X)=\frac{B(X)}{B_{\mathsf{Com}_e}(X)}$. Therefore, we obtain the following equation for the subset check:
\begin{equation*}
    [1]_1 \circ [B(\tau)]_2 = [B_{\mathsf{Com}_e}(\tau)]_2 \circ [\bar{B}(\tau)]_1
\end{equation*}

The aggregator can prove $B\subseteq\mathsf{Com}_e$ by providing the commitment $[\bar{B}(\tau)]_1$ of the quotient polynomial. This also allows us to verify the weight of $B$ via the degree check, where the aggregator should provide the degree proof $[\hat{B}(\tau)]_1$:
\begin{equation} \label{eq:deg-check-t}
    [\tau^t]_1 \circ [B(\tau)]_2 = [1]_2 \circ [\hat{B}(\tau)]_1
\end{equation}

Finally, to ensure that $B(X)$ is non-zero, a dummy party $\mathsf{P}_0$ is always required in $B$ (as was done in~\cite{garg2024threshold}), which can also be checked by a KZG opening equation
\begin{equation*}
    [1]_1 \circ \mathcolorbox{lightgray}{[B(\tau)]_2} = [\tau - 1]_2 \circ \mathcolorbox{lightgray}{[Q_0(\tau)]_1} + [1]_T.
\end{equation*}
Notice that the minimum weight of $B$ should be $t+1$ after introducing $\mathsf{P}_0$ to $B$, so we have $\hat{B}(X)=X^{t+1}B(X)$, and the first term of Equation \ref{eq:deg-check-t} should also be raised to $[\tau^{t+1}]_1$.

So far, we have replaced the Plonk-style weight proofs by introducing a new representation of $\mathsf{Com}_e$ and new polynomials with their commitments. Now the aggregated signature $\bm{w}$ consists of
\begin{equation*}
    [B(\tau)]_2,\mathsf{aPK},[Q_Z(\tau)]_1,[Q_x(\tau)]_1,[\hat{Q}_x(\tau)]_1,\sigma^*,[\hat{B}(\tau)]_1,[\bar{B}(\tau)]_1,[Q_0(\tau)]_1
\end{equation*}

which should satisfy the verification that consists of the following six pairing equations (the group elements from the signature are shaded, and the rest are public group elements):
\begin{equation} \label{eq:ste-naive}
    \begin{aligned}\
        [SK(\tau)]_1 \circ \mathcolorbox{lightgray}{[B(\tau)]_2} &= [1]_2 \circ \mathcolorbox{lightgray}{\mathsf{aPK}} + [Z(\tau)]_2 \circ \mathcolorbox{lightgray}{[Q_Z(\tau)]_1} \\&+ [\tau]_2 \circ \mathcolorbox{lightgray}{[Q_x(\tau)]_1} \\
        [\tau]_2 \circ \mathcolorbox{lightgray}{[Q_x(\tau)]_1} &= [1]_2 \circ \mathcolorbox{lightgray}{[\hat{Q}_x(\tau)]_1} \\
        \mathsf{tag} \circ \mathcolorbox{lightgray}{\mathsf{aPK}} &= [1]_1 \circ \mathcolorbox{lightgray}{\sigma^*} \\
        [\tau^{t+1}]_1 \circ \mathcolorbox{lightgray}{[B(\tau)]_2} &= [1]_2 \circ \mathcolorbox{lightgray}{[\hat{B}(\tau)]_1} \\
        [1]_1 \circ \mathcolorbox{lightgray}{[B(\tau)]_2} &= [B_{\mathsf{Com}_e}(\tau)]_2 \circ \mathcolorbox{lightgray}{[\bar{B}(\tau)]_1} \\
        [1]_1 \circ \mathcolorbox{lightgray}{[B(\tau)]_2} &= [\tau - 1]_2 \circ \mathcolorbox{lightgray}{[Q_0(\tau)]_1} + [1]_T \\
    \end{aligned}
\end{equation}

\subsection{Our Construction}
Now we give our construction based on the above discussion. Let $N$ be the maximum size of the entire set of parties. Without loss of generality, we set $|\mathbb{H}|=N+1$ where the index 0 is reserved for a dummy party with $\mathsf{sk}_0=1$ and $\mathsf{pk}_0=[1]_1$.

We use the Eq-DL proof in Section \ref{subsec:eq-dl} as a knowledge proof for $\mathsf{sk}$ at two different steps:
\begin{itemize}
    \item In $\mathsf{HintGen}$, prove that $\mathsf{sk}_i$ is the discrete logarithm of $[\mathsf{sk}_iL_i(\tau)]_1$ with respect to $[L_i(\tau)]_1$ and check it in $\mathsf{HintVerify}$. Otherwise, a malicious member can construct a hint using an existing $\mathbb{G}_1$ element (e.g., another member's public key or an element related to $\mathsf{CRS}$) without knowing its discrete logarithm, and pass the original check of $\mathsf{HintVerify}$.
    \item In $\mathsf{Sign}$, prove that $\mathsf{sk}_i$ is the discrete logarithm of $\sigma_i$ with respect to $\mathsf{tag}$ and check it in $\mathsf{PartVerify}$. Note that this is equivalent to the original BLS signature and verification while avoiding pairings in signature verification at the cost of an increased signature size.
\end{itemize}

\begin{construction}[Linear verifiable c-STS] \label{cons:c-sts-naive}
We specify each algorithm of linear verifiable c-STS as follows.
\begin{itemize}
    \item $\mathsf{Setup}(1^\lambda,N)$: Sample $\tau\gets_\$\mathbb{Z}_p$ and output
    \begin{equation*}
        \mathsf{CRS}=([1]_1,[\tau^1]_1,\dots,[\tau^{N+1}]_1,[1]_2,[\tau^1]_2,\dots,[\tau^{N+1}]_2).
    \end{equation*}

    \item $\mathsf{KeyGen}(1^\lambda)$: Sample $x\gets_\$\mathbb{Z}_p$ and output $\mathsf{pk}=[x]_1,\mathsf{sk}=x$.

    \item $\mathsf{HintGen}(\mathsf{CRS},\mathsf{sk}_i,N,i)$: Output
    \begin{equation*}
        \begin{aligned}
            \mathsf{hint}_i=\Big(&[\mathsf{sk}_iL_i(\tau)]_1,[\mathsf{sk}_i(L_i(\tau)-L_i(0))]_1,\\
            &\Big[\mathsf{sk}_i\frac{L_i^2(\tau)-L_i(\tau)}{Z(\tau)}\Big]_1,\Big[\mathsf{sk}_i\frac{L_i(\tau)-L_i(0)}{\tau}\Big]_1,\\
            &\Big\{\Big[\mathsf{sk}_i\frac{L_i(\tau)L_j(\tau)}{Z(\tau)}\Big]_1\Big\}_{j\in[N]',j\ne i},\pi^{\mathsf{hint}}_i\Big),
        \end{aligned}
    \end{equation*}
    where the Eq-DL proof
    \begin{equation*}
        \pi^{\mathsf{hint}}_i=\mathsf{EqProve}([1]_1,\mathsf{pk}_i,[L_i(\tau)]_1,[\mathsf{sk}_iL_i(\tau)]_1;\mathsf{sk}_i).
    \end{equation*}

    \item $\mathsf{Preprocess}(\mathsf{CRS},\{\mathsf{hint}_i,\mathsf{pk}_i\}_{i\in [N]})$: Verify each hint and let $V\subseteq[N]$ be the set of indices with valid hints. Set $\mathsf{sk}_0=1,\mathsf{sk}_i=0$ for $i\notin V'$. Output
    \begin{equation*}
        \begin{aligned}
            \mathsf{ak}=\Big(&V,\{\mathsf{pk}_i\}_{i\in V'},\{\mathsf{hint}_i^{(2)}\}_{i\in V'},\{\mathsf{hint}_i^{(3)}\}_{i\in V'},\\
            &\{\mathsf{hint}_i^{(4)}\}_{i\in V'},\Big\{\sum_{j\in V',j\ne i}\mathsf{hint}_{j,i}^{(5)}\Big\}_{i\in V'}\Big),\\
            \mathsf{vk}=\Big(&\sum_{i\in V'}\mathsf{hint}_i^{(1)},[Z(\tau)]_2\Big)=(C,Z).
        \end{aligned}
    \end{equation*}

    \item $\mathsf{EpochRepr}(e)$: Obtain committee $\mathsf{Com}_e\subseteq[N]$ of epoch $e$, compute $B_{\mathsf{Com}_e}(X)$ by interpolating $\{(1,1),(\omega^i,0)_{i\notin\mathsf{Com}_e'}\}$:
    \begin{equation*}
        B_{\mathsf{Com}_e}(X)=\prod_{i\notin\mathsf{Com}_e'}\frac{X-\omega^i}{1-\omega^i},
    \end{equation*}
    and output $B_e=[B_{\mathsf{Com}_e}(\tau)]_2$.

    \item $\mathsf{Sign}(\mathsf{sk}_i,m,e)$: Let $\mathsf{tag}=\mathcal{H}_2(m,e)$ and output $\sigma_i=(s_i,\pi^{\mathsf{sign}}_i)$, where
    \begin{equation*}
        \begin{aligned}
            s_i&=\mathsf{sk}_i\cdot \mathsf{tag},\\
            \pi^{\mathsf{sign}}_i&=\mathsf{EqProve}([1]_1,\mathsf{pk}_i,\mathsf{tag},s_i;\mathsf{sk}_i).
        \end{aligned}
    \end{equation*}

    \item $\mathsf{PartVerify}(\mathsf{pk}_i,\sigma_i,m,e)$: Parse $\sigma_i=(s_i,\pi^{\mathsf{sign}}_i)$. Let $\mathsf{tag}=\mathcal{H}_2(m,e)$ and output 1 iff $\mathsf{EqVrfy}([1]_1,\mathsf{pk}_i,\mathsf{tag},s_i;\pi^{\mathsf{sign}}_i)$.

    \item $\mathsf{Aggr}(\mathsf{CRS},\mathsf{ak},m,\{\sigma_i\}_{i\in S},t,e)$: Parse each $\sigma_i=(s_i,\pi^{\mathsf{sign}}_i)$. Discard signatures from those whose hints did not verify (under $\mathsf{HintVerify}$), whose signatures did not verify (under $\mathsf{PartVerify}$), and who do not belong to $\mathsf{Com}_e$. Let $S$ be the set of remaining signatures, and return $\perp$ unless $1\le t\le |\mathsf{Com}_e|$ and $|S|\ge t$. Proceed as follows:
    \begin{enumerate}[label=\arabic*.]
        \item Compute $B(X)$ by interpolating $\{(1,1),(\omega^i,0)_{i\notin S'}\}$:
        \begin{equation*}
            B(X)=\prod_{i\notin S'}\frac{X-\omega^i}{1-\omega^i},
        \end{equation*}
        and compute $B=[\sum_{i=0}^NB(\omega^i)L_i(\tau)]_2$.

        \item Compute $\mathsf{aPK}=\frac{1}{N+1}(\sum_{i\in S}B(\omega^i)\mathsf{pk}_i+[1]_1)$.

        \item Compute
        \begin{equation*}
            \begin{aligned}
               Q_Z&=[Q_Z(\tau)]_1=\sum_{i=0}^{N}B(\omega^i)\cdot(\mathsf{ak}^{(4)}_i+\mathsf{ak}^{(6)}_i),\\
               Q_x&=[Q_x(\tau)]_1=\sum_{i=0}^{N}B(\omega^i)\cdot\mathsf{ak}^{(5)}_i,\\
               \hat{Q}_x&=[Q_x(\tau)\cdot\tau]_1=\sum_{i=0}^{N}B(\omega^i)\cdot\mathsf{ak}^{(3)}_i.
           \end{aligned}
        \end{equation*}
        
        \item Compute $\sigma^*=\frac{1}{N+1}(\sum_{i\in S}B(\omega^i)s_i+\mathsf{tag})$.

        \item Compute $\hat{B}(X)=X^{t+1}B(X)$ and set $\hat{B}=[\hat{B}(\tau)]_1$.

        \item Compute $\bar{B}(X)=\prod_{i\in \mathsf{Com}_e\setminus S}\frac{X-\omega^i}{1-\omega^i}$ and set $\bar{B}=[\bar{B}(\tau)]_1$.

        \item Compute $Q_0(X)=\frac{B(X)-1}{X-1}$ and set $Q_0=[Q_0(\tau)]_1$.

        \item Output $\sigma=(B,-\mathsf{aPK},-Q_Z,-Q_x,\hat{Q}_x,\sigma^*,-\hat{B},-\bar{B},-Q_0)^\top$.
    \end{enumerate}
    
    \item $\mathsf{Verify}(\mathsf{vk},m,\sigma,t,e)$: Let $\mathsf{tag}=\mathcal{H}_2(m,e)$ and set $Z_0=[\tau-1]_2$, compute
    \begin{equation*}
        \begin{aligned}
            \bm{A} &= \begin{pmatrix}
            C & [1]_2 & Z & [\tau]_2 & 0 & 0 & 0 & 0 & 0 \\
            0 & 0 & 0 & [\tau]_2 & [1]_2 & 0 & 0 & 0 & 0 \\
            0 & \mathsf{tag} & 0 & 0 & 0 & [1]_1 & 0 & 0 & 0 \\
            [\tau^{t+1}]_1 & 0 & 0 & 0 & 0 & 0 & [1]_2 & 0 & 0 \\
            [1]_1 & 0 & 0 & 0 & 0 & 0 & 0 & B_e & 0 \\
            [1]_1 & 0 & 0 & 0 & 0 & 0 & 0 & 0 & Z_0
            \end{pmatrix}\\
            \bm{b}&=([0]_T,[0]_T,[0]_T,[0]_T,[0]_T,[1]_T)^\top
        \end{aligned}
    \end{equation*}
    and output 1 iff $\bm{A}\circ\sigma=\bm{b}$. This corresponds to Equation \ref{eq:ste-naive} since it is fully linear, i.e., each equation only consists of pairing computation, while each pairing has at least one public input (we use colors to demonstrate this).
\end{itemize}
\end{construction}

To verify the well-formedness of each hint, we define the $\mathsf{HintVerify}$ algorithm as follows.

$\mathsf{HintVerify}(\mathsf{CRS},\mathsf{hint}_i,\mathsf{pk}_i)\to 1/0$: Parse $\mathsf{hint}_i=(h_1,h_2,h_3,h_4,\\\{h_5^{(j)}\}_{j\in[N]',j\ne i},\pi^{\mathsf{hint}}_i)$. Output 1 iff $\mathsf{EqVrfy}([1]_1,\mathsf{pk}_i,[L_i(\tau)]_1,h_1;\pi^{\mathsf{hint}}_i)=1$ and
    \begin{equation*}
        \begin{aligned}
            h_1\circ[1]_2&=\mathsf{pk}_i\circ[L_i(\tau)]_2\\
            h_2\circ[1]_2&=\mathsf{pk}_i\circ[L_i(\tau)-L_i(0)]_2\\
            h_3\circ[1]_2&=\mathsf{pk}_i\circ\Big[\frac{L_i^2(\tau)-L_i(\tau)}{Z(\tau)}\Big]_2\\
            h_4\circ[1]_2&=\mathsf{pk}_i\circ\Big[\frac{L_i(\tau)-L_i(0)}{\tau}\Big]_2\\
            h_{5,j}\circ[1]_2&=\mathsf{pk}_i\circ\Big[\frac{L_i(\tau)L_j(\tau)}{Z(\tau)}\Big]_2,\forall j\in[N]'\land j\ne i
        \end{aligned}
    \end{equation*}

\begin{remark}
    We assume that the $\mathsf{Setup}$ algorithm also outputs the commitments of the Lagrange basis polynomials $[L_i(\tau)]_{1,2}$. This is because almost all steps of our scheme use these values, and pre-processing them can significantly reduce the related computational cost. Since these values can be easily computed and publicly verified by $\mathsf{CRS}$, this assumption will not affect the security of our scheme.
\end{remark}

\section{Making c-STS Practical} \label{sec:improve}
In this section, we analyze the efficiency of Construction \ref{cons:c-sts-naive}. We first review the time complexities of several fundamental polynomial operations:
\begin{itemize}
    \item Given a polynomial $f(X)$ with $f(0)\ne 0$, its multiplicative inverse modulo $X^n$ can be computed in $O(n\log n)$ time by Newton's iteration and the Fast Fourier Transform (FFT).
    \item For a polynomial $f(X)$ of degree $n$, evaluating $f(X)$ at $n$ arbitrary distinct points requires $O(n\log^2 n)$ time by a divide-and-conquer approach. Alternatively, if the evaluation points are the $N$-th roots of unity ($N\ge n$), the evaluation can be computed in $O(N\log N)$ time by FFT.
    \item Let $\omega$ be an $N$-th root of unity and $S\subseteq[N]$ with $|S|=n$. Computing the coefficients of $f_S(X)=\prod_{i\in S}(X-\omega^i)$ by divide-and-conquer requires $O(n\log^2n)$ time. However, $f_S(X)$ can also be obtained by computing the inverse of $-\frac{Z(X)}{f_S(X)}$ modulo $X^{n+1}$, which takes $O((N-n)\log^2(N-n)+n\log n)$ time and is asymptotically faster when $N-n\ll n$.
\end{itemize}

Let $n=|\mathsf{Com}_e|$ be the size of the epoch $e$'s committee and $s=|S|$ be the number of valid partial signatures; then for any valid $S$ we have $t\le s\le n$. Under the assumption that $N\gg n$, the efficiency of Construction \ref{cons:c-sts-naive} can be analyzed as follows:
\begin{itemize}
    \item The complexity of $\mathsf{EpochRepr}$ is $O(N\log N)$, since obtaining $[B_{\mathsf{Com}_e}(\tau)]_2$ requires to compute $B_{\mathsf{Com}_e}(X)=\prod_{i\in[N]\setminus\mathsf{Com}_e}\frac{X-\omega^i}{1-\omega^i}$, which has complexity $O(N\log N)$ when $N\gg n$.
    \item The complexity of $\mathsf{Aggr}$ is also $O(N\log N)$:
    \begin{enumerate}[label=\arabic*.]
        \item Computing $B(X)$ has complexity $O(N\log N)$ (similar to computing $B_{\mathsf{Com}_e}(X)$).
        \item Evaluating $B(\omega^i)_{i\in S}$ from $B(X)$ has complexity $O(N\log N)$ since $\deg(B)=N-s$.
        \item Computing $[B(\tau)]_2=[\sum_{i=0}^NB(\omega^i)L_i(\tau)]_2$ is $O(n)$ since there are only $O(n)$ non-zero $B(\omega^i)$. Similarly, the complexity of computing $aPK,\sigma^*,[Q_Z(\tau)]_1,[Q_x(\tau)]_1,[\hat{Q}_x(\tau)]_1$ is also $O(n)$.
        \item Since $\hat{B}(X)=X^{t+1}B(X)$, $[\hat{B}(\tau)]_1$ can be computed directly from the coefficients of $B(X)$, which has complexity $O(N)$.
        \item Computing $\bar{B}(X)$ has complexity $O(n\log^2n)$, and computing $[\bar{B}(\tau)]_1$ has complexity $O(n)$.
        \item Computing the coefficients of $Q_0(X)$ and $[Q_0(\tau)]_1$ has complexity $O(N)$.
    \end{enumerate}
\end{itemize}

The analysis above demonstrates that although we have successfully removed the Plonk-style proofs in Dyna-hinTS, the efficiency of Construction \ref{cons:c-sts-naive} cannot meet our requirements in large-scale systems, where the global system size is much larger than that of each epoch's committee. Therefore, we propose improvements to Construction \ref{cons:c-sts-naive} to make it practical even if $N$ is very large. Specifically, we reduce the complexity of $\mathsf{EpochRepr}$ and $\mathsf{Aggr}$ from $O(N\log N)$ to $O(n\log^2n)$, which significantly improves their efficiency when $N\gg n$. We also reduce the complexity of $\mathsf{HintGen}$ from $O(N^2)$ to $O(N)$, which is of independent interest. We describe the details of all optimizations below.

\subsection{Improve Epoch Commitment}
Recall the equation of the subset check
\begin{equation} \label{eq:sub-check-naive}
    [1]_1 \circ \mathcolorbox{lightgray}{[B(\tau)]_2} = [B_{\mathsf{Com}_e}(\tau)]_2 \circ \mathcolorbox{lightgray}{[\bar{B}(\tau)]_1}
\end{equation}

Since $B_{\mathsf{Com}_e}(X)$ has $N-n$ zeros, we must compute it in $O(N\log N)$. In contrast, its "complementary" polynomial $C_{\mathsf{Com}_e}(X)$ with respect to the vanishing polynomial $Z(X)$, i.e.,
\begin{equation*}
    C_{\mathsf{Com}_e}(X)=\prod_{i\in\mathsf{Com}_e'}(X-\omega^i),
\end{equation*}
has only $n+1$ zeros and can be computed in $O(n\log^2n)$. Thus, multiplying $C_{\mathsf{Com}_e}(\tau)$ on both sides of equation \ref{eq:sub-check-naive} we have
\begin{equation} \label{eq:sub-check}
    [C_{\mathsf{Com}_e}(\tau)]_1 \circ \mathcolorbox{lightgray}{[B(\tau)]_2} = [k\cdot Z(\tau)]_2 \circ \mathcolorbox{lightgray}{[\bar{B}(\tau)]_1}
\end{equation}
where $k$ is some constant value (we compute it below) and can be moved into the computation of $[\bar{B}(\tau)]_1$. Notice that equation \ref{eq:sub-check} is still linear because the commitment of the vanishing polynomial $[Z(\tau)]_2$ is public. This allows us to replace the output of $\mathsf{EpochRepr}(e)$ by $[C_{\mathsf{Com}_e}(\tau)]_1$ without compromising the security properties of the scheme. The modified $\bar{B}(X)$ can be computed as follows and remains $O(n\log^2n)$:
\begin{equation*}
    \begin{aligned}
        \bar{B}(X)&=\frac{C_{\mathsf{Com}_e}(X)\cdot B(X)}{Z(X)}=\frac{\prod_{i\in\mathsf{Com}_e'}(X-\omega^i)\cdot\prod_{i\notin S,i\ne 0}\frac{X-\omega^i}{1-\omega^i}}{\prod_i(X-\omega^i)}\\
        &=\frac{\prod_{i\in \mathsf{Com}_e\setminus S}(X-\omega^i)}{\prod_{i\notin S,i\ne 0}(1-\omega^i)}=\prod_{i\in\mathsf{Com}_e\setminus S}(X-\omega^i)\cdot\frac{\prod_{i\in S}(1-\omega^i)}{N+1}\\
    \end{aligned}
\end{equation*}

\subsection{Improve Signature Aggregation}
There are three steps with complexity $\Omega(N)$ in the algorithm $\mathsf{Aggr}$ of Section \ref{sec:sts-naive}, namely computing $B$, $\hat{B}$, and $Q_0$. We optimize these steps separately as follows.

\noindent\textbf{Computing $B$.}
Since the purpose of computing $B(X)$ is to obtain $\{B(\omega^i)\}_{i\in S}$, which has only $O(n)$ non-zero elements, it is possible to reduce the complexity to $O(n)$ if we can bypass interpolating $B(X)$, i.e., to compute $\{B(\omega^i)\}_{i\in S}$ without computing $B(X)$ itself.

Let $f_k(X)=\prod_{i\ne k}(X-\omega^i)=\frac{X^{N+1}-1}{X-\omega^k}$, then by $L_k(\omega^k)=1$ we have $f_k(\omega^k)=\frac{N+1}{\omega^k}$. Since $B(X)=\prod_{i\notin S,i\ne 0}\frac{X-\omega^i}{1-\omega^i}$, for each $\{\omega^k\}_{k\in S}$ we want to compute
\begin{equation*}
    B(\omega^k)=\prod_{i\notin S,i\ne 0}\frac{\omega^k-\omega^i}{1-\omega^i}        =\frac{\prod_{i\ne 0,i\ne k}\frac{\omega^k-\omega^i}{1-\omega^i}}{\prod_{i\in S,i\ne k}\frac{\omega^k-\omega^i}{1-\omega^i}}.
\end{equation*}

For the numerator, we have
\begin{equation*}
    \begin{aligned}
        \prod_{i\ne 0,i\ne k}\frac{\omega^k-\omega^i}{1-\omega^i}&=\frac{\prod_{i\ne 0,i\ne k}(\omega^k-\omega^i)}{\prod_{i\ne 0,i\ne k}(1-\omega^i)}=\frac{f_k(\omega^k)/(\omega^k-1)}{f_0(1)/(1-\omega^k)}\\
        &=\frac{N+1}{\omega^k(\omega^k-1)}\cdot\frac{1-\omega^k}{N+1}=-\omega^{-k}.
    \end{aligned}
\end{equation*}

For the denominator, let $g(X)=\prod_{i\in S}(X-\omega^i)$, then $g'(\omega^k)=\prod_{i\in S,i\ne k}(\omega^k-\omega^i)$ and
\begin{equation*}
    \prod_{i\in S,i\ne k}\frac{\omega^k-\omega^i}{1-\omega^i}=\frac{\prod_{i\in S,i\ne k}(\omega^k-\omega^i)}{\prod_{i\in S,i\ne k}(1-\omega^i)}=(1-\omega^k)\cdot\frac{g'(\omega^k)}{g(1)}.
\end{equation*}

In conclusion, we can compute $B(\omega^k)$ by
\begin{equation*}
    B(\omega^k)=\frac{g(1)}{\omega^k\cdot(\omega^k-1)\cdot g'(\omega^k)}.
\end{equation*}
Since the complexity of computing $g(1)$ is $O(n)$, the complexity of computing and evaluating $g'(X)$ on $S$ is $O(n\log^2n)$, all $B(\omega^k)$ can be computed with complexity $O(n\log^2n)$. Finally, using $\{B(\omega^k)\}_{k\in S}$ to compute $B=[\sum_{i\in S'}B(\omega^i)L_i(\tau)]_2$ has complexity $O(n)$.

\noindent\textbf{Computing $\hat{B}$.}
$\hat{B}$ can be computed in a similar way to computing $B$ since we already know that $\hat{B}(1)=1,\hat{B}(\omega^i)_{i\notin S,i\ne 0}=0$ and $\hat{B}(\omega^i)_{i\in S}=\omega^{i(t+1)}B(\omega^i)$. Since $\deg(\hat{B})=(N-s)+(t+1)\le N$ when $s>t$, computing $\hat{B}=[\hat{B}(\tau)]_1=[\sum_{i\in S'}\hat{B}(\omega^i)L_i(\tau)]_1$ has complexity $O(n)$.

The only exception is $\deg(\hat{B})=N+1$ when $s=t$, where $\hat{B}(X)$ cannot be directly represented by $L_i(X)$. Let $\hat{B}'(X)=\sum_{i\in S'}\hat{B}(\omega^i)L_i(X)$; then $\hat{B}(X)=\hat{B}'(X)$ for all $\omega^i$, which implies $Z(X)|\hat{B}(X)-\hat{B}'(X)$. Since $\deg(\hat{B}')\le N$ and $\deg(Z)=N+1$, it must be
\begin{equation*}
    \hat{B}(X)=\sum_{i\in S'}\hat{B}(\omega^i)L_i(X)+k\cdot Z(X),
\end{equation*}
where $k$ equals to the leading coefficient of $B(X)$ and can be computed as
\begin{equation*}
    k=\prod_{i\notin S,i\ne 0}\frac{1}{1-\omega^i}=\frac{g(1)}{N+1}.
\end{equation*}
Therefore, computing
\begin{equation*}
    \hat{B}=[\hat{B}(\tau)]_1=\Big[\sum_{i\in S'}\hat{B}(\omega^i)L_i(\tau)\Big]_1+\frac{g(1)}{N+1}\cdot[Z(\tau)]_1
\end{equation*}
also has complexity $O(n)$.

\noindent\textbf{Computing $Q_0$.}
The method above cannot be used again for computing $Q_0=[Q_0(\tau)]_1$ since $\deg(Q_0)=N-s-1$ and $Q_0(X)=\frac{B(X)-1}{X-1}$ has no zeros on $\mathbb{H}$, which implies that the complexity of computing $Q_0$ is $\Omega(N)$ either by computing coefficients of $Q_0(X)$ or by $[Q_0(\tau)]_1=[\sum_{i=0}^NQ_0(\omega^i)L_i(\tau)]_1$.

To address this challenge, we should review the role of $Q_0$: it is the KZG opening proof of $B(\omega^0)=1$, which ensures that the dummy party $\mathsf{P}_0\in B$. Our approach here is to \textbf{regard the 1 on the right-hand side as the evaluation of some polynomial $h(X)$ at $\omega^0$}, i.e., we prove $B(\omega^0)=h(\omega^0)$ for a polynomial $h(X)$ with $h(\omega^0)=1$ publicly verifiable. This clearly does not affect the statement we need to prove, but by carefully choosing $h(X)$, the computation of $Q_0=[Q_0(\tau)]_1$ may be significantly simplified. Since the vast majority of $\mathbb{H}$ are zeros of $B(X)$, we would like $h(X)$ to have the same property, and $h(X)=L_0(X)$ becomes the most natural choice. By defining
\begin{equation} \label{eq:q0-new}
    Q_0(X)=\frac{B(X)-L_0(X)}{X-1},
\end{equation}
we have $Q_0(\omega^i)=0$ for $i\notin S,i\ne 0$, i.e., there are only $O(n)$ non-zero $Q_0(\omega^i)$. 

For $i\in S$, we know that $Q_0(\omega^i)=\frac{B(\omega^i)}{\omega^i-1}$ which can be computed in $O(n)$, and for $Q_0(1)$ we have $Q_0(1)=B'(1)-L_0'(1)$. To compute $B'(1)$ and $L_0'(1)$, notice that for any $p(X)=\prod_ip_i(X)$ we have
\begin{equation*}
    p'(X)=p(X)\cdot\sum_i\frac{p_i'(X)}{p_i(X)}.
\end{equation*}
Since $B(X)=\prod_{i\notin S,i\ne 0}p_i(X)$ and $L_0(X)=\prod_{i\ne 0}p_i(X)$ where $p_i(X)=\frac{X-\omega^i}{1-\omega^i}$ satisfies $p_i(1)=1$ and $p_i'(1)=\frac{1}{1-\omega^i}$, we have
\begin{equation*}
    \begin{aligned}
        B'(1)&=B(1)\cdot\sum_{i\notin S,i\ne 0}\frac{p'(1)}{p(1)}=\sum_{i\notin S,i\ne 0}\frac{1}{1-\omega^i}\\
        L_0'(1)&=L_0(1)\cdot\sum_{i\ne 0}\frac{p'(1)}{p(1)}=\sum_{i\ne 0}\frac{1}{1-\omega^i}
    \end{aligned}
\end{equation*}
which concludes that
\begin{equation*}
    Q_0(1)=\sum_{i\in S}\frac{1}{\omega^i-1}
\end{equation*}
and can also be computed in $O(n)$. The result above allows us to compute $Q_0=[\sum_{i\in S'}Q_0(\omega^i)L_i(\tau)]_1$ in $O(n)$ since there are only $O(n)$ non-zero $Q_0(\omega^i)$.

Finally, the verification equation corresponding to equation \ref{eq:q0-new} is
\begin{equation*}
    [1]_1\circ\mathcolorbox{lightgray}{[B(\tau)]_2}=[\tau-1]_2\circ\mathcolorbox{lightgray}{[Q_0(\tau)]_1}+[L_0(\tau)]_1\circ[1]_2,
\end{equation*}
which remains linear.

By applying the optimizations described above, our full signature verification equations can be summarized as
\begin{equation} \label{eq:sts-eq-final}
    \begin{aligned}\
        [SK(\tau)]_1 \circ \mathcolorbox{lightgray}{[B(\tau)]_2} &= [1]_2 \circ \mathcolorbox{lightgray}{\mathsf{aPK}} + [Z(\tau)]_2 \circ \mathcolorbox{lightgray}{[Q_Z(\tau)]_1}\\
        &+ [\tau]_2 \circ \mathcolorbox{lightgray}{[Q_x(\tau)]_1} \\
        [\tau]_2 \circ \mathcolorbox{lightgray}{[Q_x(\tau)]_1} &= [1]_2 \circ \mathcolorbox{lightgray}{[\hat{Q}_x(\tau)]_1} \\
        \mathsf{tag} \circ \mathcolorbox{lightgray}{\mathsf{aPK}} &= [1]_1 \circ \mathcolorbox{lightgray}{\sigma^*} \\
        [\tau^{t+1}]_1 \circ \mathcolorbox{lightgray}{[B(\tau)]_2} &= [1]_2 \circ \mathcolorbox{lightgray}{[\hat{B}(\tau)]_1} \\
        [C_{\mathsf{Com}_e}(\tau)]_1 \circ \mathcolorbox{lightgray}{[B(\tau)]_2} &= [Z(\tau)]_2 \circ \mathcolorbox{lightgray}{[\bar{B}(\tau)]_1} \\
        [1]_1 \circ \mathcolorbox{lightgray}{[B(\tau)]_2} &= [\tau-1]_2 \circ \mathcolorbox{lightgray}{[Q_0(\tau)]_1}+[L_0(\tau)]_1 \circ [1]_2 \\
    \end{aligned}
\end{equation}
which has the complexity $O(n\log^2n)$.

\subsection{Improve Silent Setup} \label{subsec:opt-setup}
Although this work shares a common silent setup phase with~\cite{garg2024hints,garg2024threshold,kate2025dyna}, we notice that the optimizations to this phase are rarely mentioned. This is likely because the setup is one-time and trivially parallelizable; hence, its cost does not critically affect the usability of the underlying schemes.

Nevertheless, we provide some improvements to the silent setup phase in this section, which do not relate to specific threshold signatures or encryption schemes. Therefore, they can also be applied to~\cite{garg2024hints,garg2024threshold,kate2025dyna} and may be of independent interest.

The silent setup phase in these schemes consists of the $\mathsf{Setup},\mathsf{KeyGen},\\\mathsf{HintGen},\mathsf{Preprocess}$ algorithms, among which we focus on the optimization of $\mathsf{HintGen}$ since the other algorithms are hard to optimize at the mathematical level. To generate $\mathsf{hint}_i$, one must compute $H_i=(h_1,h_2,h_3,h_4,\{h_{5,j}\}_{j\in[N]',j\ne i})$ in advance, where
\begin{equation*}
    \begin{aligned}
        h_1&=[L_i(\tau)]_1, h_2=[L_i(\tau)-L_i(0)]_1,\\
        h_3&=\Big[\frac{L_i^2(\tau)-L_i(\tau)}{Z(\tau)}\Big]_1, h_4=\Big[\frac{L_i(\tau)-L_i(0)}{\tau}\Big]_1,\\
        h_{5,j}&=\Big[\frac{L_i(\tau)L_j(\tau)}{Z(\tau)}\Big]_1,\forall j\in[N]'\land j\ne i.
    \end{aligned}
\end{equation*}
In practice, each party may compute its own $H_i$ by itself, or the aggregator may compute all of them and deliver each $H_i$ to the $i$-th party. In any case, the $i$-th party multiplies its own $\mathsf{sk}_i$ with all the values in $H_i$ and sends them to the aggregator. Therefore, we only focus on how to compute $H_i$ efficiently below.

We denote the multi-scalar multiplication of $k$ $\mathbb{G}_i$ ($i=1,2$) points by $\text{msm}_i(k)$. The naive approach computes each polynomial in $H_i$ and its commitment:
\begin{itemize}
    \item $h_1=[L_i(\tau)]_1$ is already available. Since $[L_i(0)]_1=[\frac{1}{N+1}]_1$ is a constant for all $i\in[N]'$, we also consider it publicly available, hence computing $h_2$ requires 1 $\mathbb{G}_1$ addition.
    \item Computing $h_3$ and $h_4$ requires to obtain the coefficients of $h_3(X)$ and $h_4(X)$ first, and $\text{msm}_1(N)$ each for their commitments since they have degree $N-1$.
    \item Similarly, computing $h_{5,j}$ requires to obtain the coefficients of $h_{5,j}(X)$ first, and then $\text{msm}_1(N)$. These steps apply to all $j\in[N]',j\ne i$.
\end{itemize}
In conclusion, the total complexity of computing $H_i$ is $O(N^2)$, and can be reduced to $O(N^2/\log N)$ when applying a faster MSM algorithm (e.g., Pippenger’s method~\cite{pippenger1980evaluation}).

Our observation is that this complexity can be further reduced to $O(N)$. Specifically, we compute $h_3,h_4,h_5$ as follows:
\begin{itemize}
    \item By $\sum_{j\in[N]'}L_j(X)=1$, we have
    \begin{equation*}
        \sum_{j\in[N]',j\ne i}\frac{L_i(X)L_j(X)}{Z(X)}+\frac{L_i^2(X)}{Z(X)}=\frac{L_i(X)}{Z(X)},
    \end{equation*}
    namely $h_3(X)=-\sum h_{5,j}(X)$. Therefore, we can compute $h_3=-\sum h_{5,j}$ by $(N-1)\mathbb{G}_1$ additions after we obtain $h_5$.
    \item For $h_4(X)$, we have
    \begin{equation*}
        \begin{aligned}
            \frac{L_i(X)-L_i(0)}{X}&=\frac{1}{X}\Big(\frac{\omega^i}{N+1}\cdot\frac{X^{N+1}-1}{X-\omega^i}-\frac{1}{N+1}\Big)\\
            &=\frac{1}{X}\cdot\frac{\omega^i(X^{N+1}-1)-(X-\omega^i)}{(N+1)(X-\omega^i)}\\
            &=\frac{\omega^iX^N-1}{(N+1)(X-\omega^i)}\\
            &=\frac{(X^{N+1}-1)-(X^{N+1}-\omega^iX^N)}{(N+1)(X-\omega^i)}\\
            &=\frac{L_i(X)}{\omega^i}-\frac{X^N}{N+1}\\
        \end{aligned}
    \end{equation*}
    Therefore, $h_4=\frac{[L_i(\tau)]_1}{\omega^i}-\frac{[\tau^N]_1}{N+1}$ and can be computed by 2 $\mathbb{G}_1$ multiplications and 1 $\mathbb{G}_1$ addition.
    \item For $h_{5,j}(X)$, since
    \begin{equation*}
        \frac{L_i(X)L_j(X)}{Z(X)}=\frac{\omega^i\omega^j}{(N+1)^2}\cdot\frac{X^{N+1}-1}{(X-\omega^i)(X-\omega^j)},
    \end{equation*}
    we have
    \begin{equation*}
        \begin{aligned}
            h_{5,j}(\omega^i)&=\frac{1}{N+1}\cdot\frac{\omega^j}{\omega^i-\omega^j}\\
            h_{5,j}(\omega^j)&=\frac{1}{N+1}\cdot\frac{\omega^i}{\omega^j-\omega^i}
        \end{aligned}
    \end{equation*}
    and $h_{5,j}(\omega^k)=0$ for all $k\ne i,j$. Therefore,
    \begin{equation*}
        \begin{aligned}\
            [h_{5,j}(\tau)]_1&=\sum_{k=0}^Nh_{5,j}(\omega^k)[L_k(\tau)]_1\\
            &=h_{5,j}(\omega^i)[L_i(\tau)]_1+h_{5,j}(\omega^j)[L_j(\tau)]_1.
        \end{aligned}
    \end{equation*}
    Notice that $h_{5,j}(\omega^i)+h_{5,j}(\omega^j)=-\frac{1}{N+1}$, we can further write
    \begin{equation*}
        [h_{5,j}(\tau)]_1=-\frac{[L_i(\tau)]_1}{N+1}+h_{5,j}(\omega^j)([L_j(\tau)]_1-[L_i(\tau)]_1)
    \end{equation*}
    where the first term is constant across all $h_{5,j}$. This allows us to compute each $h_{5,j}$ by 1 $\mathbb{G}_1$ multiplication and 2 $\mathbb{G}_1$ additions, with a single multiplication for computing $-\frac{[L_i(\tau)]_1}{N+1}$.
\end{itemize}

In conclusion, the total complexity of our approach to compute $H_i$ is $(N+3)\mathbb{G}_1$ multiplications and $(3N+1)\mathbb{G}_1$ additions, providing a significant improvement over the naive approach.

\section{The final c-STS construction}
In this section, we propose our core contribution, i.e., the practical silent threshold
signature scheme for dynamic committees. Our final construction is demonstrated in Section \ref{subsec:construct-sts}, and we analyze its security properties in Appendix \ref{app:prf-sts}.

\subsection{Construction of Practical c-STS Scheme} \label{subsec:construct-sts}
Our final construction combines the construction in Section \ref{sec:sts-naive} and the improvements in Section \ref{sec:improve}. For clarity, differences from Construction \ref{cons:c-sts-naive} are shown in orange.

\begin{construction}[Practical c-STS] \label{cons:c-sts-final}
We specify each algorithm of practical c-STS as follows.
\begin{itemize}
    \item The $\mathsf{Setup},\mathsf{KeyGen},\mathsf{HintGen}$ and $\mathsf{Preprocess}$ algorithms are the same as Construction \ref{cons:c-sts-naive}.

    \item $\mathsf{EpochRepr}(e)$: Obtain committee $\mathsf{Com}_e\subseteq[N]$ of epoch $e$, \textcolor{orange}{compute $C_{\mathsf{Com}_e}(X)=\prod_{i\in\mathsf{Com}_e'}(X-\omega^i)$} and output $B_e=[C_{\mathsf{Com}_e}(\tau)]_1$.

    \item The $\mathsf{Sign}$ and $\mathsf{PartVerify}$ algorithms are also the same as Construction \ref{cons:c-sts-naive}.

    \item $\mathsf{Aggr}(\mathsf{CRS},\mathsf{ak},m,\{\sigma_i\}_{i\in S},t,e)$: Parse each $\sigma_i=(s_i,\pi^{\mathsf{sign}}_i)$. Discard signatures from those whose hints did not verify (under $\mathsf{HintVerify}$), whose signatures did not verify (under $\mathsf{PartVerify}$), and who do not belong to $\mathsf{Com}_e$. Let $S$ be the set of remaining signatures, and return $\perp$ unless $1\le t\le |\mathsf{Com}_e|$ and $|S|\ge t$. Proceed as follows (recall that $S'=S\cup\{0\}$):
    \begin{enumerate}[label=\arabic*.]
        \item \textcolor{orange}{Define $B(1)=1$ and $B(\omega^i)=0$ for $i\notin S'$. Compute $g(X)=\prod_{i\in S}(X-\omega^i)$ and $\{g'(\omega^i)\}_{i\in S}$. For $i\in S$, compute
        \begin{equation*}
            B(\omega^i)=\frac{g(1)}{\omega^i\cdot(\omega^i-1)\cdot g'(\omega^i)}
        \end{equation*}
        and set $B=[\sum_{i\in S'}B(\omega^i)L_i(\tau)]_2$.}

        \item Compute $\mathsf{aPK}=\frac{1}{N+1}(\sum_{i\in S'}B(\omega^i)\mathsf{pk}_i)$.

        \item Compute
        \begin{equation*}
            \begin{aligned}
               Q_Z&=[Q_Z(\tau)]_1=\sum_{i\in S'}B(\omega^i)\cdot(\mathsf{ak}^{(4)}_i+\mathsf{ak}^{(6)}_i),\\
               Q_x&=[Q_x(\tau)]_1=\sum_{i\in S'}B(\omega^i)\cdot\mathsf{ak}^{(5)}_i,\\
               \hat{Q}_x&=[Q_x(\tau)\cdot\tau]_1=\sum_{i\in S'}B(\omega^i)\cdot\mathsf{ak}^{(3)}_i.
           \end{aligned}
        \end{equation*}
        
        \item Compute $\sigma^*=\frac{1}{N+1}(\sum_{i\in S}B(\omega^i)s_i+\mathsf{tag})$.

        \item \textcolor{orange}{Let $\hat{B}(1)=1$; for $i\in S$, compute $\hat{B}(\omega^i)_{i\in S}=\omega^{i(t+1)}B(\omega^i)$ and $\hat{B'}=[\sum_{i\in S'}\hat{B}(\omega^i)L_i(\tau)]_1$. Set
        \begin{equation*}
            \hat{B}=[\hat{B}(\tau)]_1=\left\{
            \begin{aligned}\
            &\hat{B}',&\quad |S|>t \\
            &\hat{B}'+\frac{g(1)}{N+1}\cdot[Z(\tau)]_1,&\quad |S|=t
            \end{aligned}
            \right.
        \end{equation*}
        }

        \item Compute $\bar{B}(X)=\textcolor{orange}{\frac{g(1)}{N+1}\prod_{i\in \mathsf{Com}_e\setminus S}(X-\omega^i)}$ and set $\bar{B}=[\bar{B}(\tau)]_1$.

        \item \textcolor{orange}{Let $Q_0(X)=\frac{B(X)-L_0(X)}{X-1}$ (but do not compute $Q_0(X)$ and $Q_0=[Q_0(\tau)]_1$ directly). For $i\in S$, compute $Q_0(\omega^i)_{i\in S}=\frac{B(\omega^i)}{\omega^i-1}$ and $Q_0(1)=\sum_{i\in S}\frac{1}{\omega^i-1}$. Finally, set $Q_0=[Q_0(\tau)]_1=[\sum_{i\in S'}Q_0(\omega^i)L_i(\tau)]_1$.}

        \item Output $\sigma=(B,-\mathsf{aPK},-Q_Z,-Q_x,\hat{Q}_x,\sigma^*,-\hat{B},-\bar{B},-Q_0)^\top$.
    \end{enumerate}

    \item $\mathsf{Verify}(\mathsf{vk},m,\sigma,t,e)$: Let $\mathsf{tag}=\mathcal{H}_2(m,e)$, set $Z_0=[\tau-1]_2$ and \textcolor{orange}{$E=[L_0(\tau)]_T$}, compute
    \begin{equation*}
        \begin{aligned}
            \bm{A} &= \begin{pmatrix}
            C & [1]_2 & Z & [\tau]_2 & 0 & 0 & 0 & 0 & 0 \\
            0 & 0 & 0 & [\tau]_2 & [1]_2 & 0 & 0 & 0 & 0 \\
            0 & \mathsf{tag} & 0 & 0 & 0 & [1]_1 & 0 & 0 & 0 \\
            [\tau^{t+1}]_1 & 0 & 0 & 0 & 0 & 0 & [1]_2 & 0 & 0 \\
            \textcolor{orange}{B_e} & 0 & 0 & 0 & 0 & 0 & 0 & \textcolor{orange}{Z} & 0 \\
            [1]_1 & 0 & 0 & 0 & 0 & 0 & 0 & 0 & Z_0
            \end{pmatrix}\\
            \bm{b}&=([0]_T,[0]_T,[0]_T,[0]_T,[0]_T,\textcolor{orange}{E})^\top
        \end{aligned}
    \end{equation*}
    and output 1 iff $\bm{A}\circ\sigma=\bm{b}$.
\end{itemize}
\end{construction}

\subsection{Remarks on Epoch Setup and Verification Complexity} \label{rmk:epoch-repr}
In our scheme, achieving $O(1)$ complexity for signature verification intrinsically relies on the verifier having access to $B_e$. Although we assume this in Section \ref{sec:sts-def}, how to obtain $B_e$ still requires careful consideration. In fact, whether the verifier recomputes $B_e$ or verifies an additional proof provided by a third party, this step inherently requires $\Omega(n)$ due to reading $\mathsf{Com}_e$ (which is $O(n)$).

To resolve this without imposing a heavy burden on the verifier during per-message verification, we follow the paradigm established in~\cite{kate2025dyna}, i.e., decoupling the computation or verification of $B_e$ from the per-message signature verification. Specifically, trusting $B_e$ is treated as a one-time operation per epoch rather than a per-signature overhead. This can be instantiated in two ways:
\begin{itemize}
    \item At the beginning of a new epoch, the verifier computes $B_e$ itself, which requires $O(n\log^2n)$ field operations + $\text{msm}_1(n)$, and caches $B_e$ locally. For the remainder of the epoch, any signature verified against this committee takes strict $O(1)$ time. Thus, the setup cost is amortized over all signatures within the epoch.
    \item In on-chain environments, the validation of $B_e$ can be handled by a smart contract. An untrusted off-chain entity computes $B_e$ and submits it to the contract alongside a succinct proof (e.g., a KZG opening) attesting that $B_e$ correctly corresponds to $\mathsf{Com}_e$. The smart contract performs the verification once and stores $B_e$ in its state, allowing anyone to verify signatures in epoch $e$ in $O(1)$.
\end{itemize}

In summary, while the end-to-end process inherently involves an $O(n)$ step to link the dynamic committee to the commitment, our design isolates this cost to the epoch boundary. Consequently, the signature verification complexity remains $O(1)$.

\section{Definitions of c-STE}
As another contribution of this work, we formally define a new primitive, namely the silent threshold encryption scheme for dynamic committees (c-STE), in this section, and give a practical construction in Section \ref{sec:ste-cons}. Our definition adheres to the structure of~\cite{garg2024hints,garg2024threshold,kate2025dyna}. We extend silent threshold encryption to support dynamic committees, in which the authorized committee is a subset of the entire set of parties that changes dynamically over epochs, and messages encrypted for a specific epoch can be decrypted only by the corresponding committee.

\begin{definition}[c-STE] \label{def:c-ste}
A silent threshold encryption scheme for dynamic committees consists of a tuple of algorithms $\Sigma=(\mathsf{Setup,KeyGen,HintGen,Preprocess,EpochRepr,Enc,PartDec,}\\\mathsf{PartVerify,DecAggr})$ with the following syntax:

\begin{itemize}
    \item $\mathsf{CRS}\gets\mathsf{Setup}(1^\lambda,N)$: On input the security parameter $\lambda$ and an upper bound $N$ on the maximum number of users, the $\mathsf{Setup}$ algorithm outputs a common reference string $\mathsf{CRS}$.

    \item $(\mathsf{pk},\mathsf{sk})\gets\mathsf{KeyGen}(1^\lambda)$: On input the security parameter $\lambda$, the $\mathsf{KeyGen}$ algorithm outputs a key pair $(\mathsf{pk},\mathsf{sk})$.

    \item $\mathsf{hint}_i\gets\mathsf{HintGen}(\mathsf{CRS},sk,N,i)$: On input the $\mathsf{CRS}$, the secret key $sk$, the number of parties $N$, and a position $i\in[N]$, the $\mathsf{HintGen}$ algorithm outputs a hint $\mathsf{hint}_i$.

    \item $(\mathsf{ak},\mathsf{ek})\gets\mathsf{Preprocess}(\mathsf{CRS},\{\mathsf{hint}_i,\mathsf{pk}_i\}_{i\in[N]})$: On input the $\mathsf{CRS}$, all pairs $\{\mathsf{hint}_i,\mathsf{pk}_i\}_{i\in[N]}$, the $\mathsf{Preprocess}$ algorithm outputs an aggregation key $\mathsf{ak}$ and an encryption key $\mathsf{ek}$.

    \item $B_e\gets\mathsf{EpochRepr}(e)$: On input epoch $e$, the $\mathsf{EpochRepr}$ algorithm outputs a mathematical representation $B_e$ of the epoch $e$'s committee $\mathsf{Com}_e\subseteq[N]$.

    \item $\mathsf{ct}\gets\mathsf{Enc}(\mathsf{ek},\mathsf{msg},t,e)$: On input an encryption key $\mathsf{ek}$, a message $\mathsf{msg}$, a threshold $t$, and epoch $e$, the $\mathsf{Enc}$ algorithm outputs a ciphertext $\mathsf{ct}$.

    \item $\sigma\gets\mathsf{PartDec}(\mathsf{sk},\mathsf{ct},e)$: On input a secret key $\mathsf{sk}$, a ciphertext $\mathsf{ct}$, and epoch $e$, the $\mathsf{PartDec}$ algorithm outputs a partial decryption $\sigma$.

    \item $1/0\gets\mathsf{PartVerify}(\mathsf{pk},\sigma,\mathsf{ct},e)$: On input a ciphertext $\mathsf{ct}$, a partial decryption $\sigma$, a public key $\mathsf{pk}$, and epoch $e$, the $\mathsf{PartVerify}$ algorithm outputs 1 iff the partial decryption verifies and corresponds to epoch $e$.

    \item $\mathsf{msg}\gets\mathsf{DecAggr}(\mathsf{CRS},\mathsf{ak},\mathsf{ct},\{\sigma_i\}_{i\in S},t,e)$: On input the $\mathsf{CRS}$, an aggregation key $\mathsf{ak}$, a ciphertext $\mathsf{ct}$, a set of partial decryptions $\{\sigma_i\}_{i\in S}$, a threshold $t$, and epoch $e$, the $\mathsf{DecAggr}$ algorithm outputs a message $\mathsf{msg}$.
\end{itemize}
\end{definition}

Moreover, the c-STE scheme must have the following efficiency requirements:
\begin{itemize}
    \item The size of the encryption key $\mathsf{ek}$ and the ciphertext $\mathsf{ct}$ should be $O(1)$.
    \item The complexity of $\mathsf{PartDec}$ should be $O(1)$.
    \item For epoch $e$ with the committee size $n=|\mathsf{Com}_e|$, the complexity of $\mathsf{EpochRepr}$ should be $O(n\cdot\text{polylog}(n))$.
    \item Given the commitment of a committee $B_e$, the complexity of $\mathsf{Enc}$ should be $O(1)$, and the complexity of $\mathsf{DecAggr}$ should be $O(n\cdot\text{polylog}(n))$.
\end{itemize}

We also define the correctness and semantic security of c-STE through two games between a challenger and an adversary $\mathcal{A}$. We defer the formal definition to Appendix \ref{app:def-ste}.

\section{c-STE Construction} \label{sec:ste-cons}
By eliminating the Plonk-style proofs, our c-STS scheme obtains a fully linear signature verification. This allows us to leverage the compiler proposed by~\cite{garg2024threshold} to construct a c-STE scheme from our c-STS scheme, and the c-STE scheme naturally inherits all the efficiency advantages of our c-STS scheme. We propose this construction of the practical c-STE scheme in this section, and analyze its security properties in Appendix \ref{app:prf-ste}.

\subsection{Construction of Practical c-STE Scheme}
\begin{construction}[Practical c-STE] \label{cons:c-ste}
We specify each algorithm of practical c-STE as follows, and we simplify the description of the algorithms that remain unchanged from Construction \ref{cons:c-sts-final}.
\begin{itemize}
    \item The $\mathsf{Setup},\mathsf{KeyGen},\mathsf{HintGen},\mathsf{Preprocess}$ and $\mathsf{EpochRepr}$ algorithms are the same as Construction \ref{cons:c-sts-final}.
    
    \item $\mathsf{Enc}(\mathsf{ek},\mathsf{msg},t,e)$: Sample $u\gets_\$\mathbb{Z}_p$ and let $\mathsf{tag}=\mathcal{H}_2(u,e)$. Compute the matrix $\bm{A}$ and vector $\bm{b}$ as the $\mathsf{Verify}$ algorithm of Construction \ref{cons:c-sts-final}. Sample a vector $\bm{s}\gets_\$\mathbb{Z}_p^6$ and output $\mathsf{ct}=(u,\bm{s}^\top\cdot\bm{A},\bm{s}^\top\cdot\bm{b}+\mathsf{msg})$.

    \item $\mathsf{PartDec}(\mathsf{sk}_i,\mathsf{ct},e)$: Parse $\mathsf{ct}=(u,\bm{ct}_2,ct_3)$ and output $\sigma_i=\mathsf{Sign}(\mathsf{sk}_i,u,e)$ using the $\mathsf{Sign}$ algorithm of Construction \ref{cons:c-sts-final}.

    \item $\mathsf{PartVerify}(\mathsf{pk}_i,\sigma_i,\mathsf{ct},e)$: Parse $\mathsf{ct}=(u,\bm{ct}_2,ct_3)$ and output 1 iff $\mathsf{PartVerify}(\mathsf{pk}_i,\sigma_i,u,e)=1$ using the $\mathsf{PartVerify}$ algorithm of Construction \ref{cons:c-sts-final}.

    \item $\mathsf{DecAggr}(\mathsf{CRS},\mathsf{ak},\mathsf{ct},\{\sigma_i\}_{i\in S},t,e)$: Parse $\mathsf{ct}=(u,\bm{ct}_2,ct_3)$ and compute $\bm{w}=\mathsf{Aggr}(\mathsf{CRS},\mathsf{ak},u,\{\sigma_i\}_{i\in S},t,e)$ using the $\mathsf{Aggr}$ algorithm of Construction \ref{cons:c-sts-final}. If the aggregation returns $\perp$, output $\perp$. Otherwise, output $\mathsf{msg}^*=ct_3-\bm{ct}_2\circ\bm{w}$.
\end{itemize}
\end{construction}
